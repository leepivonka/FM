;==============================================================================
; 65816 ANS Forth Kernel (ITC)
; Set up to run on WDC w65c265sxb board.
; Based on http://forum.6502.org/viewtopic.php?f=9&t=8577
;	https://github.com/Martin-H1/65816/tree/main/Forth816
;
; Architecture:
;	Threading:	Indirect Threaded Code (ITC)
;	Cell size:	16-bit
;
; ITC execution model:
;	Each word's Code Field contains a pointer to the interpreter extension machine code.

; Default word entry environment:
;	Native mode, P.M=16-bit, P.X=16-bit
;	A  (16-bit) scratch
;	X  (16-bit) Parameter Stack Pointer (PSP) - index to TOS in PStack
;	Y  (16-bit) Instruction Pointer (IP)	  - points to next CFA
;	S  (16-bit) Return Stack Pointer (RSP)	  - hardware stack
;	D		Direct Page base (UP)	  points to user area
;	DB		Data Bank ($00)
;
; User area is the direct page.
; Parameter stack TOS is at PStack,x
; Return stack TOS is at 1,s
; Temp* variables are available without preserving.

Blocks = 1	; enable block code
Opt_Full = 1	; enable everything

	.p816			;select 65816 CPU
	.list on
	.listbytes unlimited ;include all object bytes in listing (since we load the listing file)
	.feature loose_string_term ; used to put quotes in strings
	.feature org_per_seg
	.smart		;watch the code to determine P_M & P_X sizes

; Memory segment layout.
; These are absolute instead of relative:
;   * absolute addresses are in listing
;   * allow loading by report-scraping the listing (no link step).
; This could be changed to be more conventional.										 
  .bss	; uninitialized read-write
	.org $300	; in 65c265sxb onboard RAM, away from Mensch monitor stuff in lo RAM
BssEnd=$6000
  .code ; initialized, read-only
	.org $6000	; in 65c265sxb onboard RAM for now


; CPU P flag bits:
P_I	= $04	; interrupt disable
P_D	= $08	; decimal
P_X	= $10	; X size
P_M	= $20	; M size

;==============================================================================
; ca65 Macros

  .macro NEXT ; Inner interpreter dispatch
	jmp Next1
	.endmacro


  .macro PUSHA ; Push value in A onto parameter stack
	DEX
	DEX
	STA PStack,x
	.endmacro


  .macro POPA ; Pop top of parameter stack into A
	INX
	INX
	LDA PStack-2,x	; set A & flags
	.endmacro


; Word header flag & length byte bits:
F_Immediate	= $80	; when compiling, execute this word
F_Hidden	= $40	; smudged
F_Inline	= $20	; code can be compiled inline
F_LengthMask	= $1F	; mask for name length bits

 .macro Header name, flags ; generate a dictionary entry header
;	name	- string literal for the name field
;	flags	- F_Immediate, F_Hidden, or 0

	.local L1,L2
;	WordHeader_Extend	; user-supplied extension
L1:	.byte name	;name of word as a string
L2:	.byte L2-L1+flags ;flags & length of name (see F_*)
	.addr DictLink ;link to previous word in dictionary chain
;		The fields above are at negative offsets from label.
DictLink .set * ; remember the entry point of this word for later
;		xt & nt of this word are here.
;label:	; CFA & nt & xt is here.  This is the address that "Search-WordList" returns.
	.endmacro

DictLink .set 0 ;ptr to nt & xt of previous word, 0=end of list
		; Used by Header to chain the generated word headers together.

;=============================================================================
 .zeropage ; Current direct page, aka User Area
; ca65 will use direct page addressing for these automatically.
; This is only a definition - it does not allocate any memory by itself - see UserFirst.

; On the 65816 with it's moveable direct page, we'll just make the direct page the user area.
; We don't put it at $000000 so we don't conflict with the '265 Mensch monitor stuff.
; The current copy of these variables should be accessed using direct addressing.
; Absolute addressing doesn't take the D register into account.
	.org 0 ; create offsets in user area segment

U_Base:		.res 2		; Numeric I/O base
U_State:	.res 2		; Compile state, 0=interp, else compile
U_DP:		.res 2		; Dictionary memory pointer
U_Latest:	.res 2		; ptr Latest word header
U_LatestXt:	.res 2		; ptr Latest word xt (see :NoName )
U_Scr:		.res 2		;
U_SourceId:	.res 2		; Source Id
U_Blk:		.res 2		; 0 or # number of the block being interpreted.
				;   https://forth-standard.org/standard/block/BLK
U_ParseBuf:	.res 2		; Source string buffer address
U_ParseLen:	.res 2		; Source string length
U_ToIn:		.res 2		; >IN  parse offset in U_ParseBuf
U_HoldIndex:	.res 2		; current start of HOLD string
U_FSP:		.res 2		; Floating-Point stack index
U_FSFull:	.res 2		; FSP full value
U_FSEmptyP1:	.res 2		; FSP empty +1 value
U_RandState:	.res 4		; random # state
;U_ExceptionHandler: .res 2	; Exception handler

; These are in the user area so they are simpler to use.
; It's OK to have these as per-user even though they don't need to be.

;-- Scratch variables, default is not preserved across subroutine calls
W:		.res 2		; (current CFA in Next1)
Temp0:		.res 2
Temp2:		.res 2
Temp4:		.res 2
Temp6:		.res 2

U_User:		.res 20	; user-defined user variable space

		.align $10 ; make prettier listing
PStack:		.res 64	; Parameter stack
PSP_Init = *-PStack	; Parameter stack initial pointer

RStack:		.res 64	; Return (65816 hardware) stack
RSP_Init = *-1	; Return stack initial pointer

UserFixedSize: ; size of fixed part of user area
Dict_Base: ; RAM per-user dictionary starts here

;==============================================================================
 .bss

TIB_Size	= $fc
TIB_Base:	.res TIB_Size	; Terminal Input Buffer

BlockAnchor:	.res 2		; BlockBuf *  anchor for block buffers
TempAbsA:	.res 2		; absolute scratch

	.align $100 ; page align direct page for best performance
UserFirst:	.res UserFixedSize ; 1st user area starts here
				; 1st user area dictionary starts here

;==============================================================================
 .code

;------------------------------------------------------------------------------
Cold: ; Cold start

	CLC
	XCE		; CPU to native mode
	rep #P_M|P_X|P_D ; 16-bit registers, not decimal

	lda #UserFirst	; Initialize User Pointer
	tcd

	LDA #RSP_Init+UserFirst
	TAS		; reset return stack
	LDX #PSP_Init	; reset parameter stack

	phk		; Set Data Bank to Code bank ($00)
	plb

	lda #10
	sta U_Base

	stz U_State	; STATE = 0

	lda #Dict_Base+UserFirst ; DP = Dict_Base
	sta U_DP

	lda #Last_ROM_Word ; LATEST = last ROM word
	sta U_Latest

	stz BlockAnchor

;	lda #'R'	;???debug
;	jsr EmitA

	lda #Quit	; Jump to QUIT (outer interpreter)
	jmp NextA


;Break1: .word *+2  ;???debug
;	NEXT

 Header "Cold",0 ; Cold start
	.word Cold

; SECTION 12: SYSTEM WORDS (QUIT, ABORT)


 Header "Bye",0 ; ( -- ) halt the system
  ; https://forth-standard.org/standard/tools/BYE
Bye:	.word $00e084	; CFA, Mensch Reset


 Header "Abort",0 ; ( -- ) reset stacks and go to QUIT
  ; https://forth-standard.org/standard/core/ABORT
Abort:	.word DoCol	; CFA
	.word PSP_Reset	; reset parm stack
	; For now, ABORT calls QUIT which resets stacks
	.word DotQuoteRun
	.word 5
	.byte "?",0," " ; this phrase makes the simulator stop copying from an input file
	.byte $0d,$0a
	.word Quit


 Header "Quit",0 ; ( -- ) outer interpreter loop
  ; Resets return stack, reads and interprets input forever
  ; https://forth-standard.org/standard/core/QUIT
Quit:	.word DoCol	; CFA
	.word RSP_Reset			; Reset return stack
	.word Zero,State,Store		; STATE= interpret
	.word Zero,Blk,Store		; Blk= =
	.word Zero,Source_Id,Store	; Source-ID= 0
@Loop:					; Begin
	.word Refill
	.word Drop		; drop Refill result
	.word Space
	.word Interpret			;   Interpret the input line
	.word State,Fetch,ZeroEq	;   if interpreting
	.word ZBranch,@8
	.word DotQuoteRun,3		;     Print " ok"
	.byte " ok"
@8:	.word CR
	.word Branch,@Loop		;  Again


; Header "RSP-Reset",0 ; ( -- ) (R: ... -- ) reset the return stack pointer
RSP_Reset: .word *+2
	LDA #RSP_Init+UserFirst	; S= UserFirst.RSP_Init
	TAS
	NEXT


 Header "Refill",0 ; ( -- flag )  Attempt to fill the input buffer from the input source
  ; https://forth-standard.org/standard/core/REFILL
  ; https://forth-standard.org/standard/block/REFILL
Refill: .word DoCol
	.word Lit,TIB_Base		;   Use terminal buffer
	.word Dup,ParseBuf,Store
	.word Lit,TIB_Size		;   push buf length
	.word Accept			;   Read line
	.word ParseLen,Store		;   save line size
	.word Zero,ToIn,Store		;   init parse offset
	.word True	; return success
	.word Exit

  ; https://forth-standard.org/standard/core/SAVE-INPUT
Save_Input: .word *+2
	pei (U_ParseBuf)	; save existing source

	pei (U_ToIn)
	pei (U_SourceId)

 Header "Restore-Input",0 ; ( )

Restore_Input: .word *+2
	pla
	sta U_Blk
	pla

	pla

	NEXT


 Header "Evaluate",0 ; ( addr len -- )  interpret a string
  ; https://forth-standard.org/standard/core/EVALUATE
Evaluate: .word DoCol
	.word Save_Input	; save existing source
	.word Zero,Blk,Store
	.word ParseLen,Store	; set string as source
	.word ParseBuf,Store
	.word Zero,ToIn,Store
	.word Lit,$ffff,Source_Id,Store
	.word Interpret		; interpret it
	.word Restore_Input	; restore saved source
	.word Exit

  .endif ; Blocks


 Header "Interpret",0 ; ( -- ) parse and execute/compile words from input
  ; https://forth-standard.org/standard/usage#usage:command
Interpret: .word DoCol	; CFA
@next_word:
	.word Parse_Name ; Parse next space-delimited word
	.word Dup,ZBranch,@done	; Check for empty word (length = 0)

	.word TwoDup		; ( caddr len caddr len )
	.word Latest,Fetch	; search for the word in dictionary
	.word Search_WordList	; ( addr len wid -- 0 | xt 1 | xt -1 )
	.word Dup,ZBranch,@not_found ; result: 0=not found, 1=normal, -1=immediate
	.word ZeroGt,ZBranch,@exec_word

	.word State,Fetch,ZBranch,@exec_word ; executing or compiling?

	; Compile the word: , the xt
	.word CompileComma
	.word TwoDrop		; 2Drop extra string
	.word Branch,@next_word

@exec_word: ; Execute: load xt, jump through code field
	.word Nip,Nip		; discard extra string copy
	.word CheckStack	; ???debug
	.word Execute		; run the xt
	.word Branch,@next_word

@not_found: ; not a name, try a number
	.word Drop		; Drop search result
	.word TwoDup		; ( caddr len caddr len )
	.word Number		; Try to convert as number
				; ( adr len -- 0 ) or ( adr len -- n -1 ) or ( adr len -- d -2 )
	.word OnePlus,Dup,ZBranch,@Single
	.word OnePlus,    ZBranch,@Double
	.word Type		; string
	.word DotQuoteRun,11
	.byte " not found "
	.word Abort

@Double: ; it's a double
	.word State,Fetch,ZBranch,@DNum
	
	.word TwoLiteral
	.word TwoDrop	; discard string
	.word Branch,@next_word

@DNum:
	.word TwoNip		; discard string, keep d
	.word Branch,@next_word

@Single: ; it's a single
	.word Drop		; drop number flag
	.word State,Fetch,ZBranch,@SNum
	
	.word Literal		; compile n
	.word TwoDrop		; 2Drop string
	.word Branch,@next_word

@SNum:
	.word Nip,Nip		; discard string, keep n
	.word Branch,@next_word

@done:
	.word TwoDrop		; discard empty parse string
	.word Exit


 Header "?Stack",0 ; ( -- ) check stack bounds

	cpx #PSP_Init+1
	bcs @CS_Err
	NEXT

@CS_Err:

	; .word Type
	jsr Abort+2

OpenBracket: .word *+2

	NEXT


 Header "]",0 ; ( -- )  switch State to compile

CloseBracket: .word *+2

	NEXT


 Header "Code",0 ; ( "name" -- )  Start a machine code word

Code:	.word DoCol

SemiCode: .word DoCol

 Header ":NoName",0 ; ( -- xt ) start compiling a new colon word, with no header
  ; https://forth-standard.org/standard/core/ColonNONAME
ColonNoname: .word DoCol
	.word Here		; remember xt
	.word Dup,LatestXt,Store
	.word Branch,Colon3


 Header ":",0 ; ( "name" -- )  start compiling a new colon word
  ; https://forth-standard.org/standard/core/Colon
Colon:	.word DoCol
	.word HeaderComma
Colon3:	; smudge
	.word Lit,DoCol,Comma	; compile CFA
	.word CloseBracket	; State= compile
	.word Exit


 Header ";",F_Immediate ;  finish compiling a new colon word
  ; https://forth-standard.org/standard/core/Semi
Semicolon: .word DoCol
	.word Lit,Exit,CompileComma	; compile Exit

	; smudge

	.word OpenBracket	; STATE= interpret
	.word Exit


 Header "Dump",0 ; ( caddr len -- )  Dump memory in hex
  ; https://forth-standard.org/standard/tools/DUMP
Dump:	.word DoCol
	.word Zero		; line length
@line:
	.word CR
	.word @Adr,DotHex	; print addr
@byte:	.word Space	;   print space
	.word @Get,CDotHex	;   print byte
	.word Over,ZeroGt,ZBranch,@done	; done?
	.word Dup,Lit,$f,And1,ZBranch,@line	; new line?
	.word Branch,@byte
	
@done:	.word ThreeDrop
	.word Exit

@Adr:	.word *+2
	lda PStack+4,x	; get caddr
	jmp PushA
	
@Get:	.word *+2
	lda (PStack+4,x) ; get byte
	inc PStack+4,x	; addr++
	dec PStack+2,x	; len--
	inc PStack+0,x	; line position++
	jmp PushA


 Header "SeeLatest",0 ; ( -- )  show code of latest word
SeeLatest: .word DoCol
	.word LatestXt,Fetch
	.word Here,Over,Minus
	.word Dump
	.word Exit


 Header ".S",0 ; ( -- ) print stack contents non-destructively
  ; https://forth-standard.org/standard/tools/DotS
DotS:	.word DoCol
	.word Lit,'<',Emit
	.word Depth,Dup,Dot
	.word Lit,'>',Emit
	.word Space
@Loop:	.word Dup,ZBranch,@End
	.word Dup,Pick,Dot
	.word OneMinus
	.word Branch,@Loop
@End:	.word Drop
	.word CR
	.word Exit

; Block storage stuff

;------- RAM disk for testing -----------------------------
RamDiskBuf = $10000 ; use bank 1

 Header "RamDiskWrite",0 ; ( adr block -- stat )  Write a 512 byte block
RamDiskWrite: .word *+2
	phy		; save IP
	jsr RamDiskWriteSub
	ply		; restore IP
	NEXT
	
RamDiskWriteSub:
	lda PStack+0,x
	cmp #$80		; check block #
	bcs @Abort
	xba			; << 9
	asl a
	and #$fe00
	tay
	phx
	lda PStack+2,x
	tax
	lda #512-1
	mvn #0,#RamDiskBuf>>16
	plx
	inx
	inx
	stz PStack,x
	rts

@Abort:	jsr Abort+2


 Header "RamDiskRead",0 ; ( adr block -- stat )  read a 512 byte block
RamDiskRead: .word *+2
	phy		; save IP
	jsr RamDiskReadSub
	ply		; restore IP
	NEXT
	
RamDiskReadSub:
	lda PStack+0,x
	cmp #$80		; check block #
	bcs @Abort
	xba			; << 9
	asl a
	and #$fe00
	ldy PStack+2,x
	phx
	tax
	lda #512-1
	mvn #RamDiskBuf>>16,#0
	plx
	inx
	inx
	stz PStack,x
	rts

@Abort:	jsr Abort+2


	.struct BlockBuf ; mass-storage block buffer
Next	.word		; next BlockBuf in chain (must be 1st in struct)
Number	.word		; block # ($ffff=not valid)
Status	.word		; b15=modified
Data	.res 1024	; FORTH data buffer (2*512 device blocks)
	.endstruct

BlkFind: ; ( u -- u )
	lda #BlockAnchor	; remember ptr to next
	sta Temp0
	lda BlockAnchor
	beq @None
@11:	tay
	lda PStack,x		; correct?
	cmp a:BlockBuf::Number,y
	beq @Found
	sty Temp0		; remember last
	lda a:BlockBuf::Next,y	; goto next in chain
	bne @11

	; not found.  recycle the last block in chain
	sta (Temp0)		; previous block is now last in chain
	lda BlockAnchor		; this block is now 1st in chain
	sta a:BlockBuf::Next,y
	sty BlockAnchor

	bra Flush1

@Found:	rts

@None:	jsr Abort+2


Flush1: ; Y=BlockBuf *  write back 1 block if needed
	lda a:BlockBuf::Status,y ; need to write?
	bpl @9

	phy
	jsr TwoDup
	asl PStack+0,x		; device block #
	lda 1,s			; buf addr
	clc
	adc #BlockBuf::Data
	sta PStack+2,x

	jsr TwoDup
	jsr RamDiskWriteSub ; ( adr block -- stat )  Write 1st device block
	inx
	inx
	lda PStack-2,x
	bne @Abort

	inc PStack+0,x
	clc
	lda PStack+2,x
	adc #512
	sta PStack+2,x
	jsr RamDiskWriteSub ; ( adr block -- stat )  Write 2nd dvice block
	inx
	inx
	lda PStack-2,x
	bne @Abort

	ply
@9:	rts

@Abort:	ply
	jsr Abort+2


 Header "BlockAdd",0 ; ( u -- )  add blocks to system
BlockAdd: .word @18

@11:
	phy		; save IP
	ldy U_DP		; remember start
	tya
	clc
	adc #.sizeof(BlockBuf)	; allot space
	sta U_DP
	lda BlockAnchor		; link into chain
	sta a:BlockBuf::Next,y
	sty BlockAnchor
	lda #$ffff		; invalid block #
	sta a:BlockBuf::Number,y
	lda #0			; not modified
	sta a:BlockBuf::Status,y
	ply		; restore IP

@18:	dec PStack,x		; while u>0
	bpl @11
	inx
	inx
	NEXT


 Header "Block",0 ; ( u -- a-addr ) return address of block buffer filled with block u.
  ; https://forth-standard.org/standard/block/BLOCK
Block: .word *+2
	phy		; save IP
	jsr BlkFind		; find a BlockBuf
	lda PStack,x
	cmp a:BlockBuf::Number,y
	beq @ok
	sta a:BlockBuf::Number,y
	lda #0
	sta a:BlockBuf::Status,y

	phy
	jsr TwoDupSub
	asl PStack+0,x
	clc
	lda 1,s
	adc #BlockBuf::Data+0
	sta PStack+2,x
	jsr TwoDupSub		; read 1st device block
	jsr RamDiskReadSub
	inx			; check status
	inx
	lda PStack,x
	bne @Abort
	inc PStack+0,x		; read 2nd device block
	clc
	lda PStack+2,x
	adc #512
	sta PStack+2,x
	jsr RamDiskReadSub
	inx			; check status
	inx
	lda PStack+0,x
	bne @Abort
	ply

	lda #0
	sta a:BlockBuf::Status,y
@ok:	sty PStack,x
	ply		; restore IP
	NEXT

@Abort: ply
	jsr Abort+2
	

 Header "Buffer",0 ; ( u -- a-addr ) return address of block buffer assigned to block u.
  ; https://forth-standard.org/standard/block/BUFFER
Buffer: .word *+2
	phy		; save IP
	jsr BlkFind	; find a block
	lda PStack,x
	sta a:BlockBuf::Number,y
	lda #0
	sta a:BlockBuf::Status,y
	sty PStack,x
	ply		; restore IP
	NEXT
	

 Header "Empty-Buffers",0 ; ( -- ) Unassign all block buffers. No disk writes. 
  ; https://forth-standard.org/standard/block/EMPTY-BUFFERS
Empty_Buffers: .word *+2
	phy		; save IP
	ldy BlockAnchor
	beq @19
@11:	lda #$ffff
	sta a:BlockBuf::Number,y
	lda #0
	sta a:BlockBuf::Status,y
	lda a:BlockBuf::Next,y
	bne @11
@19:	ply		; restore IP
	NEXT


 Header "Flush",0 ; ( -- )
  ; https://forth-standard.org/standard/block/FLUSH
Flush: .word DoCol

	.word Empty_Buffers

 Header "Save-Buffers",0 ; ( -- ) Save UPDATEd block buffers. Mark as unmodified
  ; https://forth-standard.org/standard/block/SAVE-BUFFERS
Save_Buffers: .word *+2
	ply		; save IP
	ldy BlockAnchor
	beq @19
@11:	jsr Flush1
	lda a:BlockBuf::Next,y
	bne @11
@19:	ply		; restore IP
	NEXT


 Header "Update",0 ; ( -- ) Mark the current block buffer as modified. 
  ; https://forth-standard.org/standard/block/UPDATE

	ldy BlockAnchor

@19:	ply		; restore IP

 Header "Parse",0 ; ( "name" c -- addr u )  "Parse input with delimiter character"
  ; https://forth-standard.org/standard/core/PARSE
  ; Find word in input string delimited by character given. Do not skip leading delimiters
  ; -- this is the main difference to PARSE-NAME.
  ; ANSI discussion: http://www.forth200x.org/documents/html3/rationale.html#rat:core:PARSE
Parse:	.word *+2	; CFA
	dex			; alloc 2nd parm cell
	dex
	phy			; save IP
	sep #P_M

	ldy U_ToIn
	phy			;save start index

	dey			;span word
@22:	lda PStack+2,x		; load delimiter
@21:	iny			;  step to next char
	cpy U_ParseLen		;  at end-of-string?
	bcs @28
	cmp (U_ParseBuf),y	;  delimiter?
	beq @Found
	cmp #' '		; we're searching for whitespace?
	bne @21
	lda (U_ParseBuf),y	; is it whitespace?
	cmp #' '
	bcs @22

@Found:	iny			;  skip delimiter
	sty U_ToIn
	dey
	bra @29
@28:	sty U_ToIn
@29:
	rep #P_M

	tya		; calc length
	sec
	sbc 1,s
	sta PStack+0,x

	pla		; calc start addr
	clc
	adc U_ParseBuf
	sta PStack+2,x

	ply		; restore IP
	NEXT

 Header "Number",0 ; ( adr len -- 0 ) or ( adr len -- n -1 ) or ( adr len -- d -2 ) or ( adr len -- fp -3 )
  ; Converts the text beginning at adr for len, with regard to Base, to a binary
  ; value that is single-length if no valid punctuation occurs, and
  ; double-length if valid punctuation does occur.  The string may contain
  ; a preceding negative sign;

@Neg	= 1	; negative sign found flag

	inx

	cmp W		; compare with n

	tay

 Header "Search-WordList",0 ; ( caddr u wid -- 0 | xt 1 | xt -1)  "Search for a word in a wordlist"
  ; Find the definition named in the string.
  ; If the definition is not found, return zero.
  ; If the definition is found, return its execution token xt.
  ; If the definition is immediate, also return one (1), otherwise also return minus-one (-1).
  ; https://forth-standard.org/standard/search/SEARCH_WORDLIST
Search_WordList: .word *+2
@HdrPtr	= Temp0  ; ptr to current word Header (-@Off)
@Quick	= Temp2  ; pattern last char & length
@StartY = Temp4  ; char compare start offset
@PatPtr = Temp6  ; ptr to pattern addr
@Off = $83 ; Chosen so 8-bit Y turns negative as we step off the end of the name chars.
		; @Off is used because the 65816 doesn't do negative indexes.
	phy		; save IP

	lda #@Off-3-1	; char compare start offset
	sec
	sbc PStack+2,x
	sta @StartY

	lda PStack+4,x	; ptr to pattern
	clc
	sbc @StartY
	sta @PatPtr

	lda PStack+2-1,x ; get pattern name length in hi byte
	sep #P_M
	ldy #@Off-4	; get pattern name last char in lo byte
	lda (@PatPtr),y
	rep #P_M
	sta @Quick	; save @Quick compare value

	lda PStack+0,x	; A= wid (wordlist head pointer)
	bra @WCur

@WNxt:	rep #P_M|P_X
	ldy @HdrPtr	; Y= header pointer
@WNxtY:	lda a:@Off-2,y	; follow link to next header
@WCur:	beq @nfnd	;   end of chain?
	sec		; apply offset for indexing
	sbc #@Off
	tay
	lda a:@Off-4,y	; check length & smudge & last char
	eor @Quick
	and #F_LengthMask<<8+F_Hidden<<8+$df
	bne @WNxtY

	sty @HdrPtr	; save header pointer
	sep #P_M|P_X
	ldy @StartY	; for each char in name
@NNxt:	iny
	bmi @fnd
	lda (@HdrPtr),y	;   compare char
	eor (@PatPtr),y
	beq @NNxt	;   exact match?
	and #$df	;   more difference than a case mismatch?
	bne @WNxt
	lda (@HdrPtr),y	;   is it a letter?
	and #$df
	sec
	sbc #'A'
	cmp #'Z'-'A'+1
	bcc @NNxt
	bra @WNxt

@fnd:			;found the word
	rep #P_M|P_X
	ldy @HdrPtr	; Y= header pointer
	lda a:@Off-3-1,y ; set flag from F_Immediate
	ora #$7fff
	bmi @fnd2
	lda #1
@fnd2:	sta PStack+2,x
	tya		; set xt
	clc
	adc #@Off
	sta PStack+4,x
	bra @99

@nfnd:			;word not found
	stz PStack+4,x	; set flag
	inx		; drop len
	inx
@99:	ply		; restore IP
	jmp Drop+2	; drop wid

  ; Remove u. Rotate u+1 items on the top of the stack. An ambiguous condition

  ; https://forth-standard.org/standard/core/ROLL

	inx
	lda PStack-2,x
	beq @99

	asl a		; X+= 2*u
	adc W
	tax

	sta PStack,x

	cpx W
	bne @12
	sty PStack,x	; store xu
	ply		; restore IP

  .endif

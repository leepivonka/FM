;==============================================================================
; 65816 ANS Forth Kernel (STC)
; Set up to run on WDC w65c265sxb boarddiff
; Based on http://forum.6502.org/viewtopic.php?f=9&t=8577
;
; Architecture:
;	Threading:	Subroutine Threaded Code (STC)
;	Cell size:	16-bit
;
; Default register conventions:
;	A  (16-bit) Working register / scratch
;	X  (16-bit) Parameter Stack Pointer (PSP) - points to TOS in direct page
;	Y  (16-bit) scratch
;	S  (16-bit) Return Stack Pointer (RSP)	  - hardware stack
;	D		Direct Page base (UP)	  points to user area
;	DB		Data Bank ($00)
;	Native mode, P.M=16-bit, P.X=16-bit
;
; STC execution model:
;	Each word consists of machine code.
;	Calls done with JSR, exits done with RTS.



	.p816			;select 65816 CPU
	.list on
	.listbytes unlimited ;include all object bytes in listing (since we load the listing file)
	.feature loose_string_term ; used to put quotes in strings
	.feature org_per_seg
	.smart		;watch the code to determine sizes
	.i16
	.a16

  .bss	; uninitialized RAM
	.org $300	; away from '265 Mensch monitor stuff in lo RAM
  .code ; ROM
	.org $6000	; in RAM for now


; CPU P flag bits:
P_I	= $04	; interrupt disable
P_D	= $08	; decimal
P_X	= $10	; X size
P_M	= $20	; M size

;==============================================================================
; ca65 Macros

  .macro PUSH ; Push value in A onto parameter stack
	DEX
	DEX
	STA 0,X
	.endmacro


  .macro POP ; Pop top of parameter stack into A
	LDA 0,X
	INX
	INX
	.endmacro


; Word header flag & length byte bits:
F_Immediate	= $80
F_Hidden	= $40	; smudged
F_LengthMask	= $1F

 .macro Header name, flags ; generate a dictionary entry header
;	name	- string literal for the name field
;	flags	- F_Immediate, F_Hidden, or 0

	.local L1,L2
;	WordHeader_Extend	; user-supplied extension
L1:	.byte name	;name of word as a string
L2:	.byte L2-L1+flags ;flags & length of name (see F_*)
	.addr DictLink ;link to previous word in dictionary chain
;		The fields above are at negative offsets from label.
DictLink .set * ; remember the entry point of this word for later
;		xt & nt of this word are here.
;label:	; machine code is here.  This is the address that "Search-WordList" returns.
	.endmacro

DictLink .set 0 ;ptr to nt & xt of previous word, 0=end of list
		; Used by Header to chain the generated word headers together.

;=============================================================================
 .zeropage ; Direct Page, aka User Area
; ca65 will use direct page addressing for these automatically.
; On the 65816 with it's moveable direct page, we'll just make the direct page the user area.
; We don't put it at $000000 so we don't conflict with the '265 Mensch monitor stuff.
; The current copy of these variables should be accessed using direct addressing.
; Absolute addressing doesn't take the D register into account.
	.org 0 ; create offsets in user area segment

U_Base:		.res 2		; Numeric I/O base
U_State:	.res 2		; Compile state, 0=interp, else compile
U_DP:		.res 2		; Dictionary memory pointer
U_Latest:	.res 2		; ptr Latest word header
U_ParseBuf:	.res 2		; Source string buffer address
U_ParseLen:	.res 2		; Source string length
U_ToIn:		.res 2		; >IN  parse offset in U_ParseBuf
;U_ExceptionHandler: .res 2	; Exception handler

; These are in the user area so they are simpler to use.
; It's OK to have these as per-user even though they don't need to be.

;-- Scratch variables, default is not preserved across subroutine calls
W:			.res 2
Temp0:		.res 2
Temp2:		.res 2
Temp4:		.res 2
Temp6:		.res 2

ParmStack:	.res 64
PSP_INIT:		; Parameter stack initial pointer

User_Size: ; size of fixed part of user area

;==============================================================================
 .bss ; unitialized RAM

;TempAbsA:	.res 2		; absolute scratch

TIB_SIZE	= $fe
TIB_Base:	.res TIB_SIZE	; Terminal Input Buffer

	.align $100 ; page align direct page for best performance
UserFirst:	.res User_Size	; 1st user area starts here

ReturnStack:	.res 64
RSP_INIT	= *-1		; Return stack initial pointer (hardware)

Dict_Base: ; RAM dictionary start


;==============================================================================
 .code ; ROM

;------------------------------------------------------------------------------
FORTH_INIT: ; SYSTEM INITIALIZATION

	CLC
	XCE		; Clear emulation bit → native mode

	REP #P_M|P_X|P_D ; 16-bit registers, not decimal

	lda #UserFirst	; Initialize User Pointer
	tcd

	phk		; Set Data Bank to Code bank ($00)
	plb

	LDX #PSP_INIT	; Parameter stack pointer
	LDA #RSP_INIT
	TAS		; Hardware (return) stack pointer

	jsr Decimal

	stz U_State	; User area: STATE = 0

	lda #Dict_Base	; User area: DP = Dict_Base
	sta U_DP

	lda #Last_ROM_Word ; User area: LATEST = last ROM word
	sta U_Latest

;	lda #'R'	;???debug
;	jsr EmitA

	jmp Quit	; Jump to QUIT (outer interpreter)

; SECTION 1: PARAMETER STACK

 Header "PSP-Reset",0 ; ( ... -- )  clear parameter stack
PSP_Reset:
	ldx #PSP_INIT
	rts

 Header "Dup",0 ; ( a -- a a )
  ; https://forth-standard.org/standard/core/DUP
Dup:	LDA 0,X		; Load TOS
PushA:	DEX
	DEX
	STA 0,X		; Push A
	rts


 Header "Drop", 0 ; ( a -- )
  ; https://forth-standard.org/standard/core/DROP
Drop:	INX
	INX
	rts


 Header "Swap", 0 ; ( a b -- b a )
  ; https://forth-standard.org/standard/core/SWAP
Swap:	lda 0,X		; b (TOS)
	ldy 2,X		; a (NOS)
	sty 0,X		; TOS = a
	sta 2,X		; NOS = b
	rts


 Header "Over", 0 ; ( a b -- a b a )
  ; https://forth-standard.org/standard/core/OVER
Over:	LDA 2,X		; a (NOS)
	bra PushA	; Push copy of a


 Header "Rot", 0 ; ( a b c -- b c a )
  ; https://forth-standard.org/standard/core/ROT
Rot:	ldy 4,X		; a (bottom)
	LDA 2,X		; b
	STA 4,X		; bottom slot = b
	LDA 0,X		; c (TOS)
	STA 2,X		; middle slot = c
	sty 0,X		; TOS = a
	rts


 Header "Nip",0 ; ( a b -- b )
  ; https://forth-standard.org/standard/core/NIP
Nip:	LDA 0,X		; b (TOS)
NipA:	INX
	INX
	STA 0,X		; Overwrite a with b
	rts


 Header "Tuck",0 ; ( a b -- b a b )
  ; https://forth-standard.org/standard/core/TUCK
Tuck:	jsr Swap
	bra PushA


 Header "2Drop", 0 ; ( a b -- )
  ; https://forth-standard.org/standard/core/TwoDROP
TwoDrop:
	INX
	INX
	bra Drop

ThreeDrop:

	inx
	bra TwoDrop	; Drop b a


 Header "2Dup", 0 ; ( a b -- a b a b )
  ; https://forth-standard.org/standard/core/TwoDUP
TwoDup:	ldy 0,x		; b
	lda 2,x		; a
PushYA:	dex
	dex
	dex
	dex
	sta 2,x
	sty 0,x
	rts


 Header "2Over", 0 ; ( a b c d -- a b c d a b )
  ; https://forth-standard.org/standard/core/TwoOVER
TwoOver: ldy 4,x	; a
	lda 6,x		; b
	bra PushYA


 Header "2Swap", 0 ; ( a b c d -- c d a b )
  ; https://forth-standard.org/standard/core/TwoSWAP
TwoSwap:
	lda 0,x		; swap b & d
	ldy 4,x
	sta 4,x
	sty 0,x
	lda 2,x		; swap a & c
	ldy 6,x
	sta 6,x
	sty 2,x
	rts


 Header "Depth", 0 ; ( -- n ) number of items on parameter stack
  ; https://forth-standard.org/standard/core/DEPTH
Depth:	stx W		; save PSP
	lda #PSP_INIT
	sec
	sbc W
	LSR A		; Divide by 2 (to get cells)
	jmp PushA


 Header "Pick",0 ; ( xu...x1 x0 u -- xu...x1 x0 xu )
  ; https://forth-standard.org/standard/core/PICK
Pick:	LDA 0,X		; u
	ASL A		; * 2 (cell size)
	phx		; save PSP
	adc 1,s		; X+=A
	tax
	lda 2,x		; get cell
	plx		; restore PSP
	sta 0,x		; save result
	rts

	INX
	INX
	ply

	rts

RFrom:	ply

 Header "R@", 0 ; ( -- a ) (R: a -- a )  get a copy of top of return stack

RFetch:	lda 3,s

Negate: LDA 0,X
Negat3:	EOR #$FFFF

	rts


 Header "Abs",0 ; ( n -- |n| )

DNegate:

	lda #0

	sta 0,x
	rts

	beq @done
@loop:	asl 2,x		; shift a
	dey
	bne @loop
@done:	inx		; Drop u
	inx
	rts


 Header "RShift", 0 ; ( a u -- a>>u ) logical shift right

	inx
	rts

 Header "S>D",0 ; ( n -- d )  Convert the signed number n to the double-cell number d
  ; https://forth-standard.org/standard/core/StoD
SToD:	lda 0,x		; get n
	and #$8000	; only sign bit
	beq @3
	lda #$ffff
@3:	jmp PushA


 Header "UM*", 0 ; ( u1 u2 -- ud ) unsigned 16x16 -> 32-bit result
  ; Result: TOS = high cell, NOS = low cell
  ; https://forth-standard.org/standard/core/UMTimes
UMStar: lda #0		; Clear result.hi
	clc
	ldy #16		; for 16 bits
@loop:	ror 2,x		;   shift multiplier & result.lo
	bcc @skip	;   if bit set
	clc		;     add multiplicand to result.hi
	adc 0,x
@skip:	ror a		;   shift result.hi
	dey		;  next bit
	bne @loop
	ror 2,x		; finish result.lo
	sta 0,x		; save result.hi
	rts


 Header "M*",0 ; ( a b -- dc ) 16x16 -> 32 signed
  ; https://forth-standard.org/standard/core/MTimes
MStar:	lda 0,x		; save result sign
	eor 2,x
	php
	jsr Abs		; abs(b)
	jsr Swap	; abs(a)
	jsr Abs
	jsr UMStar	; multiply
	plp		; fix result sign
	bpl @9
	jmp DNegate
@9:	rts


 Header "*", 0 ; ( a b -- a*b ) 16x16 -> 16 (low word)
  ; https://forth-standard.org/standard/core/Times
Star:	jsr UMStar
	inx		; drop hi word
	inx
	rts


 Header "UM/MOD", 0 ; ( ud u -- ur uq ) unsigned 32/16 -> 16 remainder, 16 quotient
  ; Stack: NOS_HI=ud_high NOS=ud_low TOS=u (divisor)
  ; Quotient overflow gives strange results.
  ; This is a standard 32/16 non-restoring division
  ; https://forth-standard.org/standard/core/UMDivMOD
UMSlashMod:
	lda 2,x		; A= dividend.hi
	ldy 4,x		; move dividend.lo
	sty 2,x
	clc
	ldy #16		; for 16 bits
@loop:	rol 2,x		;   Shift quotient & dividend.lo
	rol a		;   shift dividend.hi
	bcs @sub
	cmp 0,x		;   will it fit?
	bcc @skip
@sub:	sbc 0,x		;     subtract divisor
	sec
@skip:	dey		;  next bit
	bne @loop
	rol 2,x		; finish quotient
	sta 4,x		; save remainder
	inx		; drop divisor
	inx
	rts


 Header "SM/Rem",0 ; ( d1 n1 -- n_remainder n_quotient )  Symmetric signed division
  ; Divide d_dividend by n_divisor, giving the symmetric quotient n_quotient and
  ; the remainder n_remainder. Input and output stack arguments are signed.
  ; An ambiguous condition exists if n_divisor is zero or if the quotient
  ; lies outside the range of a single-cell signed integer.
  ; http://forth-standard.org/standard/core/SMDivREM
SMSRem:	jsr ToR		; save divisor
	lda 0,x		; save dividend sign
	pha
	jsr DAbs	; make dividend positive
	lda 3,s
	jsr PushA	; get abs(divisor)
	jsr Abs
	jsr UMSlashMod	; unsigned divide

	lda 3,s		; correct quotient sign
	eor 1,s
	bpl @5
	jsr Negate
@5:

	pla		; correct remainder sign
	bpl @6
	lda #0
	sec
	sbc 2,x
	sta 2,x
@6:
	pla		; drop saved divisor
	rts


 Header "/MOD", 0 ; ( n1 n2 -- rem quot ) signed division
  ; Divide n1 by n2, giving the single-cell remainder n3 and the single-cell quotient n4. An
  ; ambiguous condition exists if n2 is zero. If n1 and n2 differ in sign, the implementation defined
  ; result returned will be the same as that returned by either the phrase >R S>D R>
  ; FM/MOD or the phrase >R S>D R> SM/REM.
  ; https://forth-standard.org/standard/core/DivMOD
SlashMod:
	jsr ToR
	jsr SToD
	jsr RFrom
	bra SMSRem


 Header "/", 0 ; ( n1 n2 -- quot ) signed division
  ; https://forth-standard.org/standard/core/Div
Slash:	jsr SlashMod
	jmp Nip		; remainder


 Header "Mod", 0 ; ( n1 n2 -- rem ) signed remainder
  ; https://forth-standard.org/standard/core/MOD
Mod:	jsr SlashMod
	inx		; Drop quotient
	inx
	rts

; SECTION 6: MEMORY


 Header "@",0 ; ( addr -- val ) fetch cell
  ; https://forth-standard.org/standard/core/Fetch
Fetch:
	lda (0,x)
	sta 0,x
	rts


 Header "!", 0 ; ( val addr -- ) store cell
  ; https://forth-standard.org/standard/core/Store
Store:
	lda 2,x		; get val
	sta (0,x)	; store at addr
	jmp TwoDrop	; drop addr val


 Header "C@", 0 ; ( addr -- byte ) fetch byte
  ; https://forth-standard.org/standard/core/CFetch
CFetch:
	sep #P_M
	lda (0,x)
	rep #P_M
	and #$ff	; zero extend
	sta 0,x
	rts


 Header "C!", 0 ; ( byte addr -- ) store byte
  ; https://forth-standard.org/standard/core/CStore
CStore:
	LDA 2,X		; get byte
	sep #P_M	; store at addr
	sta (0,x)
	rep #P_M
	jmp TwoDrop	; drop addr byte, NEXT


 Header "2@", 0 ; ( addr -- d ) fetch double cell
  ; (hi cell at addr, lo cell at addr+2, cells stored little-endian)
  ; https://forth-standard.org/standard/core/TwoFetch
TwoFetch:
	ldy 0,x		; addr
	lda a:2,y	; lo cell
	sta 0,x
	lda a:0,y	; hi cell
	jmp PushA


 Header "2!", 0 ; ( d addr -- ) store double cell
  ; https://forth-standard.org/standard/core/TwoStore
TwoStore:
	ldy 0,x		; addr
	lda 2,x		; hi cell
	sta a:0,y
	lda 0,x		; lo cell
	sta a:2,y
	jmp ThreeDrop	; 3Drop addr d


; Header "+!",ha_Inline ; ( n|u a-addr -- )  Add n | u to the single-cell number at a-addr
  ; https://forth-standard.org/standard/core/PlusStore


; Header "?",0 ; ( a-addr -- )  Display the cell value stored at a-addr.
  ; https://forth-standard.org/standard/tools/q


 Header "CMove", 0 ; ( src dst u -- ) copy u bytes from src to dst ascending
  ; https://forth-standard.org/standard/string/CMOVE
CMove:
	lda 0,x		; u (byte count)
	beq @9		; Zero count = no-op
	dec a		; decrement byte count for mvn
	phx		; save PSP
	ldy 4,x		; prepare src
	phy
	ldy 2,x		; Y=dst
	plx		; X=src
	mvn #0,#0	; block move
	plx		; restore PSP
@9:	jmp ThreeDrop	; Drop u dst src

@Run: ; ( limit index -- ) (R: -- limit index) runtime for DO

	jsr CompileCommaA
	jmp JmpComma	; compile jmp back


@Run: ; ( -- ) (R: limit index -- | limit index+1)  runtime for LOOP
	lda 3,s		; get index
	ina
	cmp 5,s		; limit == index+1?
	beq DoLoopDone	; Loop finished?
	sta 3,s		; Put updated index back
	rts		; falls into jmp back

DoLoopDone:
	ply		; save rts addr
	pla		; RDrop index
	pla		; RDrop limit
	iny		; Skip jmp back
	iny
	iny
	phy		; put rts addr back
	rts

	jsr CompileCommaA
	jmp JmpComma	; compile jmp back


; (R: limit index -- | limit index+n)  runtime for +LOOP
@Run:	lda 0,x		; pop step
	inx
	inx
	tay		; save a copy for later
	clc
	adc 3,s		; add step
	sta 3,s

	; Check if we crossed limit
	; Positive step: done when index >= limit
	; Negative step: done when index <= limit
	; Simpler check: done when (index-limit) XOR (new_index-limit) sign differs
	sec
	tya		; get step again
	bmi @Neg
	lda 3,s		; A= index-limit
	sbc 5,s
@cmp:	bvc @cmp3
	eor #$ffff	; correct sign
@cmp3:	bpl DoLoopDone
	rts		; fall into jmp back

@Neg:	lda 5,s		; A= limit-index
	sbc 3,s
	bra @cmp

 Header "Search-WordList",0 ; ( caddr u wid -- 0 | xt 1 | xt -1)  "Search for a word in a wordlist"
  ; https://forth-standard.org/standard/search/SEARCH_WORDLIST
  ; Find the definition named in the string.
  ; If the definition is not found, return zero.
  ; If the definition is found, return its execution token xt.
  ; If the definition is immediate, also return one (1), otherwise also return minus-one (-1).
Search_WordList:
@HdrPtr	= Temp0  ; ptr to current word Header (-@Off)
@Quick	= Temp2  ; pattern last char & length
@StartY = Temp4  ; char compare start offset
@PatPtr = Temp6  ; ptr to pattern addr
@Off = $83 ; Chosen so 8-bit Y turns negative as we step off the end of the name chars.
		; @Off is used because the 65816 doesn't do negative indexes.
	lda #@Off-3-1	; char compare start offset
	sec
	sbc 2,x
	sta @StartY

	lda 4,x		; ptr to pattern
	clc
	sbc @StartY
	sta @PatPtr

	sep #P_M	; build @Quick
	lda 2,x		;   get pattern length
	xba		;     in hi byte
	ldy #@Off-4	; get pattern last char
	lda (@PatPtr),y
	rep #P_M
	sta @Quick

	lda 0,x		; A= wid (wordlist head pointer)
	bra @WCur

@WNxt:	rep #P_M|P_X
	ldy @HdrPtr
@WNxtY:	lda a:@Off-2,y		;follow link to next header
@WCur:	beq @nfnd
	sec
	sbc #@Off
	tay
	lda a:@Off-4,y		;check length & part of last char
	eor @Quick
	and #F_LengthMask<<8+F_Hidden<<8+$df
	bne @WNxtY

	sty @HdrPtr
	sep #P_M|P_X
	ldy @StartY
@NNxt:	iny
	bmi @fnd
	lda (@HdrPtr),y
	eor (@PatPtr),y
	beq @NNxt
	and #$df
	bne @WNxt
	lda (@HdrPtr),y
	and #$df
	sec
	sbc #'A'
	cmp #'Z'-'A'+1
	bcc @NNxt
	bra @WNxt

@fnd:			;found the word
	rep #P_M|P_X
	ldy #@Off-3-1	; set flag
	lda (@HdrPtr),y
	ora #$7fff
	bmi @fnd2
	lda #1
@fnd2:	sta 2,x
	lda @HdrPtr	; set xt
	clc
	adc #@Off
	sta 4,x
	bra @99

@nfnd:			;word not found
	stz 4,x		; set flag
	inx		; drop len
	inx
@99:	inx		; drop wid
	inx
	rts

	clc		; skip over chars
	adc 2,x
	dea		; update rts addr
	sta 3,s
	rts


 Header 'S"',F_Immediate ; ( -- caddr len )  create a string literal
  ; https://forth-standard.org/standard/core/Sq
SQuote: jsr ParseQuote		; get string
	lda U_State
	beq @interp		; if compiling

	lda #GetInlineString
	jsr CompileCommaA
	jmp StringComma

@interp: lda 0,x
	pha			; save length
	pei (U_DP)		; save addr
	jsr Here
	jsr Swap
	lda 0,x
	jsr AllotA
	jsr CMove		; copy string
	jsr RFrom
	jsr RFrom	; stack caddr & len
	rts

GetInlineString: ; ( -- caddr len )
	ply		; get rts addr
	lda a:1,y	; get length
	iny		; skip over length
	iny
	iny
	dex		; push caddr
	dex
	sty 0,x 
	dex		; push len
	dex
	sta 0,x
	clc		; skip over chars
	adc 2,x
	dea
	pha
	rts


 Header "MustBeCompiling",0 ; ( -- ) make sure we're compiling
MustBeCompiling:
	lda U_State
	bne @9
	jsr Abort
@9:	rts 


 Header 'Abort"', F_Immediate ; If f is true, print string & abort
  ;  abort-quote  https://forth-standard.org/standard/core/ABORTq
AbortQ: jsr ParseQuote
	jsr MustBeCompiling
	lda #AbortQRun
	jsr CompileCommaA
	jsr StringComma
	rts

AbortQRun: ; ( f -- ) runtime code ------
	lda 0,x		; get f
	bne @Abort
	inx		; Drop f
	inx
	ply		; skip string
	tya
	sec
	adc a:1,y
	ina
	pha
	rts

@Abort:	ply
	phy
	iny
	lda a:0,y	; get string length
	iny		; get string addr
	iny
	sty 0,x		; push strin addr
	dex		; push string length
	dex
	sta 0,x

	jsr Type
	jmp Abort

	phd		; get user area base
	clc		; add offset
	adc 1,s
	ply		; RDrop base
	jmp PushA	; Push A

 Header "Latest",0 ; ( -- addr ) address of LATEST variable in user area
Latest:	lda #U_Latest
	bra PushAU

 Header "Base",0 ; ( -- addr ) address of BASE variable
Base:	lda #U_Base
	bra PushAU

 Header "State",0 ; ( -- addr ) address of STATE variable
State:	lda #U_State
	bra PushAU

 Header ">In",0 ; ( -- addr ) address of >IN variable
ToIn:	lda #U_ToIn
	bra PushAU

;	bra PushAU

;	bra PushAU

 Header "Source",0 ; ( -- addr len ) current input source
  ; https://forth-standard.org/standard/core/SOURCE
Source: lda U_ParseBuf

 Header "Decimal",0 ; ( -- )  set base to 10
  ; https://forth-standard.org/standard/core/DECIMAL
Decimal: lda #10		; User area: BASE = 10
Decim3:	sta U_Base
	rts

 Header "Hex",0 ; ( -- )  set base to 16

Hex:	lda #$10	; User area: BASE = 16

 Header "Count",0 ; ( addr -- addr+1 len ) counted string to addr/len
  ; https://forth-standard.org/standard/core/COUNT
Count:	lda (0,x)	; get length byte
	and #$ff
	inc 0,x
	jmp PushA


 .if 0
 Header "Bounds",0 ; ( addr u -- addr+u addr ) Prepare address for looping
  ; http://www.complang.tuwien.ac.at/forth/gforth/Docs-html/Memory-Blocks.html
Bounds:	lda 0,x
	ldy 2,x
	clc
	adc 2,x

; Header "-Trailing",0 ; ( addr n1 -- addr n2 )  trim trailing spaces

; Header "Upper",0 ; ( addr len -- )  Convert chars to uppercase

 Header "(",F_Immediate ; (  ccc(paren)  -- )  Comment
  ; https://forth-standard.org/standard/file/p
  ; http://forth-standard.org/standard/core/p
Comment:
	lda #')'
	jsr ParseA
	jmp TwoDrop


 Header "\",F_Immediate ; ( -- )  Eat remainder of parse line as a comment
  ; https://forth-standard.org/standard/core/bs
BSlash:	lda U_ParseLen
	sta U_ToIn
	rts


 Header "Parse",0 ; ( "name" c -- addr u )  "Parse input with delimiter character"
  ; https://forth-standard.org/standard/core/PARSE
  ; Find word in input string delimited by character given. Do not skip leading delimiters
  ; -- this is the main difference to PARSE-NAME.
  ; ANSI discussion: http://www.forth200x.org/documents/html3/rationale.html#rat:core:PARSE
Parse:	bra Parse2

ParseQuote: lda #'"'
ParseA:	jsr PushA
Parse2:	dex			; alloc 2nd parm cell
	dex
	sep #P_M

	ldy U_ToIn
	phy			;save start index

	dey			;span word
@22:	lda 2,x			; load delimiter
@21:	iny			;  step to next char
	cpy U_ParseLen		;  at end-of-string?
	bcs @28
	cmp (U_ParseBuf),y	;  delimiter?
	beq @Found
	cmp #' '		; we're searching for whitespace?
	bne @21
	lda (U_ParseBuf),y	; is it whitespace?
	cmp #' '
	bcs @22

@Found:	iny			;  skip delimiter
	sty U_ToIn
	dey
	bra @29
@28:	sty U_ToIn
@29:
	rep #P_M

	tya			; calc length
	sec
	sbc 1,s
	sta 0,x

	pla			; calc start addr
	clc
	adc U_ParseBuf
	sta 2,x

	rts


 Header "Parse-Name",0 ; (  (spaces)name(space)  -- c-addr u )  Skip leading spaces. Parse space delimited name.
  ; https://forth-standard.org/standard/core/PARSE-NAME
  ; c-addr is the address of the selected string within the input buffer and u is its length in
  ; characters. If the parse area is empty or contains only white space, the resulting string
  ; has length zero.
Parse_Name:
	jsr SkipWhite		;skip leading whitespace
	lda #' '
	jsr ParseA		;get the name
;	bra SkipWhite		;skip trailing whitespace


; Header "SkipWhite",0 ; ( -- ) skip whitespace on source
SkipWhite:
	sep #P_M
	ldy U_ToIn
	dey
@11:	iny
	cpy U_ParseLen
	bcs @19
	lda (U_ParseBuf),y
	cmp #' '+1
	bcc @11
@19:	sty U_ToIn
	rep #P_M
	rts


 Header "Number",0 ; ( adr len -- 0 ) or ( adr len -- n -1 ) or ( adr len -- d -2 ) or ( adr len -- fp -3 )
  ; Converts the text beginning at adr for len, with regard to Base, to a binary
  ; value that is single-length if no valid punctuation occurs, and
  ; double-length if valid punctuation does occur.  The string may contain
  ; a preceding negative sign.
Number:	stz W		; Accumulator= 0
	stz Temp0	; negative sign flag

	jsr @Get	; get next char
	bmi @num_err
	cmp #'+'	;  plus sign?
	beq @loop
	cmp #'-'	;  minus sign?
	bne @2
	dec Temp0	;  set sign flag

@loop:	jsr @Get	; get next char
	bmi @done
@2:
	sec		; get ASCII digit value
	sbc #'0'
	bcc @num_err
	CMP #10
	bcc @ok
	sbc #7
	and #$df
	CMP #10
	bcc @num_err
@ok:	CMP U_Base
	bcs @num_err

	sta Temp2	; WB = W * BASE + digit
	ldy U_Base
	lda #0
@mul2:	clc
	adc W
	dey
	bne @mul2	
	adc Temp2
	sta W
	bra @loop

@done:	LDA W
	bit Temp0	; get sign flag
	bpl @d4
	eor #$ffff	;  negate
	ina
@d4:
	sta 2,x		; Replace addr with number
	lda #.loword(-1) ; Success
	sta 0,x
	rts

@num_err:
	stz 2,x		; return fail
	inx		; Drop len
	inx
	rts

@Get: ; get next char
	dec 0,x		; more chars?
	bmi @9
	lda (2,x)	; get char
	inc 2,x
	and #$ff
@9:	rts

 Header "Bye",0 ; ( -- ) halt the system
  ; https://forth-standard.org/standard/tools/BYE
Bye:	jmp $00e084	; Mensch Reset


 Header "Abort",0 ; ( -- ) reset stacks and go to QUIT
  ; https://forth-standard.org/standard/core/ABORT
Abort:	jsr PSP_Reset	; reset parm stack
	jsr DotQuoteRun
	.word 5
	.byte "?",0," " ; this phrase makes the simulator stop copying from an input file
	.byte $0d,$0a
	bra Quit


 Header "Quit", 0 ; ( -- ) outer interpreter loop
  ; Resets return stack, reads and interprets input forever
  ; https://forth-standard.org/standard/core/QUIT
Quit:	lda #RSP_INIT			; Reset return stack
	tcs
	stz U_State			; STATE= interpret

@Loop:	lda #TIB_Base			; Use terminal buffer
	sta U_ParseBuf
	ldy #TIB_SIZE			; push buf length
	jsr PushYA
	jsr Accept			; Read line
	lda 0,x
	inx
	inx
	sta U_ParseLen			; save line size
	jsr Space			; type a space after the line
	stz U_ToIn			; init parse offset
	jsr Interpret			; Interpret the input line
	lda U_State			; STATE == 0 (interpret mode)?
	bne @Loop			; Loop back (compiling: no prompt)
	jsr DotQuoteRun
	.word 5				; Print " ok"
	.byte " ok",$0d,$0a
	bra @Loop


 Header "Interpret",0 ; ( -- ) parse and execute/compile words from input
Interpret:
@next_word:
	jsr Parse_Name		; Parse next space-delimited word
	lda 0,x
	beq @done		; no more words in source?

	jsr TwoDup		; ( caddr len caddr len )
	lda U_Latest		; search for the word in dictionary
	jsr PushA
	jsr Search_WordList	; ( addr len wid -- 0 | xt 1 | xt -1 )
	lda 0,x
	beq @not_found		; result: 0=not found, 1=normal, -1=immediate
	bmi @exec_word

	lda U_State
	beq @exec_word

	; Compile the word: , the xt
	inx			; drop flag
	inx
	jsr CompileComma
	jsr TwoDrop		; 2Drop extra string
	bra @next_word

@exec_word: ; Execute: load xt, jump through code field
	inx			; drop flag
	inx
	jsr Nip
	jsr Nip			; discard extra string copy
	jsr Execute		; run the xt
	bra @next_word

@not_found:
	inx			; Drop search result
	inx
	jsr TwoDup		; ( caddr len caddr len )
	jsr Number		; Try to convert as number
	lda 0,x
	beq @Err		; ( adr len -- 0 ) or ( adr len -- n -1 ) or ( adr len -- d -2 )
	inx
	inx
	lda U_State
	beq @nf9
	jsr Literal		; compile n
	jsr TwoDrop		; 2Drop string
	bra @next_word

@nf9:	jsr Nip
	jsr Nip			; discard string, keep n
	bra @next_word

@Err:	inx			; number flag
	inx
	jsr Type		; string
	jsr DotQuoteRun
	.word 11
	.byte " not found "
	jsr Abort

@done:	jmp TwoDrop		; discard empty parse string


 Header "[",F_Immediate ; ( -- )  switch State to interpret
  ; https://forth-standard.org/standard/core/Bracket
OpenBracket:
	stz U_State
	rts


 Header "]",0 ; ( -- )  switch State to compile
 ; https://forth-standard.org/standard/right-bracket
CloseBracket:
	stx U_State	; set to non-zero
	rts


 Header ":", 0 ; ( "name" -- )  start compiling a new colon word
  ; https://forth-standard.org/standard/core/Colon
Colon:	jsr Parse_Name	; get name  ( caddr len )
	jsr HeaderComma
	; smudge
	jmp CloseBracket ; State= compile


 Header ";",F_Immediate ;  finish compiling a new colon word
  ; https://forth-standard.org/standard/core/Semi
Semicolon:
	jsr Exit		; compile Exit

	; smudge

	jmp OpenBracket		; STATE= interpret


 Header "Dump",0 ; ( caddr len -- )  Dump memory in hex
Dump:	dec 0,x		; so we stop on zero length remaining
@line:	jsr CR
	jsr Over	; print addr
	jsr DotHex
	ldy #16		; for 16 bytes
@byte:	phy
	jsr Space	;   print space
	lda (2,x)	;   get byte
	inc 2,x
	dec 0,x
	jsr CDotHexA	;   print byte
	ply		;  next
	dey
	bne @byte
	lda 0,x		; until done
	bpl @line
	jmp TwoDrop


 Header ".S",0 ; ( -- ) print stack contents non-destructively
  ; https://forth-standard.org/standard/tools/DotS
DotS:	lda #'<'
	jsr EmitA
	jsr Depth
	jsr Dup
	jsr Dot
	lda #'>'
	jsr EmitA
	jsr Space

@loop:	lda 0,x		; while entrys remain
	beq @done
	jsr PushA	;   get
	jsr Pick
	jsr Dot		;   print
	dec 0,x
	bpl @loop	;  next
@done:	inx		; drop n
	inx
	jmp CR

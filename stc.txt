;==============================================================================
; 65816 ANS Forth Kernel (STC)
; Set up to run on WDC w65c265sxb board.
; Based on http://forum.6502.org/viewtopic.php?f=9&t=8577
;	https://github.com/Martin-H1/65816/tree/main/Forth816
;
; Architecture:
;	Threading:	Subroutine Threaded Code (STC)
;	Cell size:	16-bit
;
; STC execution model:
;	Each word consists of machine code.
;	Calls done with JSR, exits done with RTS.

; Default word entry environment:
;	Native mode, P.M=16-bit, P.X=16-bit
;	A  (16-bit) scratch
;	X  (16-bit) Parameter Stack Pointer (PSP) - index to TOS in PStack
;	Y  (16-bit) scratch
;	S  (16-bit) Return Stack Pointer (RSP)	  - hardware stack
;	D		Direct Page base (UP)	  points to user area
;	DB		Data Bank ($00)
;
; User area is the direct page.
; Parameter stack TOS is at PStack,x
; Return stack TOS is at 1,s
; Temp* variables are available without preserving.


	.p816			;select 65816 CPU
	.list on
	.listbytes unlimited ;include all object bytes in listing (since we load the listing file)
	.feature loose_string_term ; used to put quotes in strings
	.feature org_per_seg
	.smart		;watch the code to determine P_M & P_X sizes

; Memory segment layout.
; These are absolute instead of relative:
;   * absolute addresses are in listing
;   * allow loading by report-scraping the listing (no link step).
; This could be changed to be more conventional.										 
  .bss	; uninitialized read-write
	.org $300	; in 65c265sxb onboard RAM, away from Mensch monitor stuff in lo RAM
  .code ; initialized, read-only
	.org $6000	; in 65c265sxb onboard RAM for now


; CPU P flag bits:
P_I	= $04	; interrupt disable
P_D	= $08	; decimal
P_X	= $10	; X size
P_M	= $20	; M size

;==============================================================================
; ca65 Macros

  .macro PUSHA ; Push value in A onto parameter stack
	DEX
	DEX
	STA PStack,x
	.endmacro


  .macro POPA ; Pop top of parameter stack into A
	INX
	INX
	LDA PStack-2,x	; set A & flags
	.endmacro


 .macro INL_End ; end-of-code for inline compile
	rts
	.endmacro


; Word header flag & length byte bits:
F_Immediate	= $80	; when compiling, execute this word
F_Hidden	= $40	; smudged
F_Inline	= $20	; code can be compiled inline
F_LengthMask	= $1F	; mask for name length bits

 .macro Header name, flags ; generate a dictionary entry header
;	name	- string literal for the name field
;	flags	- F_Immediate, F_Hidden, or 0

	.local L1,L2
;	WordHeader_Extend	; user-supplied extension
L1:	.byte name	;name of word as a string
L2:	.byte L2-L1+flags ;flags & length of name (see F_*)
	.addr DictLink ;link to previous word in dictionary chain
;		The fields above are at negative offsets from label.
DictLink .set * ; remember the entry point of this word for later
;		xt & nt of this word are here.
;label:	; machine code is here.  This is the address that "Search-WordList" returns.
	.endmacro

DictLink .set 0 ;ptr to nt & xt of previous word, 0=end of list
		; Used by Header to chain the generated word headers together.

;=============================================================================
 .zeropage ; Current direct page, aka User Area
; ca65 will use direct page addressing for these automatically.
; This is only a definition - it does not allocate any memory by itself - see UserFirst.

; On the 65816 with it's moveable direct page, we'll just make the direct page the user area.
; We don't put it at $000000 so we don't conflict with the '265 Mensch monitor stuff.
; The current copy of these variables should be accessed using direct addressing.
; Absolute addressing doesn't take the D register into account.
	.org 0 ; create offsets in user area segment

U_Base:		.res 2		; Numeric I/O base
U_State:	.res 2		; Compile state, 0=interp, else compile
U_DP:		.res 2		; Dictionary memory pointer
U_Latest:	.res 2		; ptr Latest word header
;U_SourceId:	.res 2		; Source Id
U_ParseBuf:	.res 2		; Source string buffer address
U_ParseLen:	.res 2		; Source string length
U_ToIn:		.res 2		; >IN  parse offset in U_ParseBuf
U_HoldIndex:	.res 2		; current start of HOLD string
U_FSP:		.res 2		; Floating-Point stack index
U_FSFull:	.res 2		; FSP full value
U_FSEmptyP1:	.res 2		; FSP empty +1 value
U_RandState:	.res 4		; random # state
;U_ExceptionHandler: .res 2	; Exception handler

; These are in the user area so they are simpler to use.
; It's OK to have these as per-user even though they don't need to be.

;-- Scratch variables, default is not preserved across subroutine calls
W:		.res 2
Temp0:		.res 2
Temp2:		.res 2
Temp4:		.res 2
Temp6:		.res 2

U_User:		.res 12		; user-defined user variable space

;		.align $10 ; make prettier listing
PStack:		.res 64	; Parameter stack
PSP_Init = *-PStack	; Parameter stack initial pointer

RStack:		.res 64	; Return (65816 hardware) stack
RSP_Init = *-1	; Return stack initial pointer

UserFixedSize: ; size of fixed part of user area
Dict_Base: ; RAM per-user dictionary starts here

;==============================================================================
 .bss

TIB_Size	= $fe
TIB_Base:	.res TIB_Size	; Terminal Input Buffer

;TempAbsA:	.res 2		; absolute scratch

	.align $100 ; page align direct page for best performance
UserFirst:	.res UserFixedSize ; 1st user area starts here
				; 1st user area dictionary starts here

;==============================================================================
 .code

;------------------------------------------------------------------------------
Cold: ; Cold start

	CLC
	XCE		; CPU to native mode
	rep #P_M|P_X|P_D ; 16-bit registers, not decimal

	lda #UserFirst	; Initialize User Pointer
	tcd

	LDA #RSP_Init+UserFirst
	TAS		; reset return stack
	LDX #PSP_Init	; reset parameter stack

	phk		; Set Data Bank to Code bank ($00)
	plb

	jsr Decimal

	stz U_State	; STATE = 0

	lda #Dict_Base+UserFirst ; DP = Dict_Base
	sta U_DP

	lda #Last_ROM_Word ; LATEST = last ROM word
	sta U_Latest

;	lda #'R'	;???debug
;	jsr EmitA

	jmp Quit	; Jump to QUIT (outer interpreter)


 Header "Cold",0 ; Cold start
	bra Cold

;==============================================================================
; SECTION 1: PARAMETER STACK

 Header "PSP-Reset",0 ; ( ... -- )  clear parameter stack
PSP_Reset:
	ldx #PSP_Init
	rts


 .if 0
 Header "0",0 ; ( -- 0 )  push 0
  ; Used frequently, so define this instead of using literals.
Zero:
	dex
	dex
	stz PStack,x
	rts
  .endif


 Header "Dup",0 ; ( x -- x x )
  ; https://forth-standard.org/standard/core/DUP
Dup:
	LDA PStack,X	; Load TOS
PushA:	DEX		; Push A
	DEX
	STA PStack,x
	rts


 Header "?Dup",0 ; ( x -- 0 | x x )
  ; https://forth-standard.org/standard/core/qDUP
QDup:	lda PStack,x
	bne PushA
	rts


 Header "Drop",F_Inline ; ( x -- )
  ; https://forth-standard.org/standard/core/DROP
Drop:
	INX
	INX
	INL_End


 Header "Swap",0 ; ( x1 x2 -- x2 x1 )
  ; https://forth-standard.org/standard/core/SWAP
Swap:
	lda PStack+0,X	; x2
	ldy PStack+2,X	; x1
	sty PStack+0,X
	sta PStack+2,X
	rts


 Header "Over",0 ; ( x1 x2 -- x1 x2 x1 )
  ; https://forth-standard.org/standard/core/OVER
Over:
	LDA PStack+2,X	; x1
	bra PushA


 Header "Rot",0 ; ( x1 x2 x3 -- x2 x3 x1 )
  ; https://forth-standard.org/standard/core/ROT
Rot:
	ldy PStack+4,X	; x1
	LDA PStack+2,X	; x2
	STA PStack+4,X
	LDA PStack+0,X	; x3
	STA PStack+2,X
	sty PStack+0,X
	rts


 Header "Tuck",0 ; ( x1 x2 -- x2 x1 x2 )
  ; https://forth-standard.org/standard/core/TUCK
Tuck:
	jsr Swap
	bra PushA	; New TOS = x2


 Header "2Drop",0 ; ( x1 x2 -- )
  ; https://forth-standard.org/standard/core/TwoDROP
TwoDrop:
	INX
	INX
	bra Drop


; Header "3Drop",0 ; ( x1 x2 x3 -- )
ThreeDrop:
	inx		; Drop x2
	inx
	bra TwoDrop	; Drop x2 x1


 Header "2Dup",0 ; ( a b -- a b a b )
  ; https://forth-standard.org/standard/core/TwoDUP
TwoDup:
	ldy PStack+0,x	; b
	lda PStack+2,x	; a
PushYA:	dex
	dex
	dex
	dex
PutYA:	sty PStack+0,x
	sta PStack+2,x
	rts


 Header "2Nip",0 ; ( da db -- db )
  ; like https://forth-standard.org/standard/core/NIP
TwoNip:
	lda PStack+2,x	; pop YA
	ldy PStack+0,x
TwoNipYA: inx
	inx
	inx
	inx
	bra PutYA	; Overwrite da


 Header "2Over",0 ; ( a b c d -- a b c d a b )
  ; https://forth-standard.org/standard/core/TwoOVER
TwoOver:
	lda PStack+6,x	; b
	ldy PStack+4,x	; a
	bra PushYA


 Header "2Swap",0 ; ( a b c d -- c d a b )
  ; https://forth-standard.org/standard/core/TwoSWAP
TwoSwap:
	lda PStack+0,x	; swap b & d
	ldy PStack+4,x
	sta PStack+4,x
	sty PStack+0,x
	lda PStack+2,x	; swap a & c
	ldy PStack+6,x
	sta PStack+6,x
	sty PStack+2,x
	rts


 Header "Depth",0 ; ( -- n ) number of items on parameter stack
  ; https://forth-standard.org/standard/core/DEPTH
Depth:
	stx W		; save PSP
	lda #PSP_Init
	sec
	sbc W
	LSR A		; Divide by 2 (to get cells)
	jmp PushA


 Header "Pick",0 ; ( xu...x1 x0 u -- xu...x1 x0 xu )
  ; https://forth-standard.org/standard/core/PICK
Pick:
	LDA PStack+0,X	; u
	ASL A		; * 2 (cell size)
	phx		; save PSP
	adc 1,s		; X+=A
	tax
	lda PStack+2,x	; get cell
	plx		; restore PSP
	sta PStack+0,x	; save result
	rts


 .if 0
 Header "Roll",0 ; ( xu xu-1 ... x0 u -- xu-1 ... x0 xu )
  ; Remove u. Rotate u+1 items on the top of the stack. An ambiguous condition
  ; exists if there are less than u+2 items on the stack before ROLL is executed.
  ; https://forth-standard.org/standard/core/ROLL
Roll:
	inx		; pop u
	inx
	lda PStack,x
	beq @99

	stx W		; remember PSP
	asl a		; X+= 2*u
	adc W
	tax

	ldy PStack,x	; save xu
@12:	lda PStack-2,x	; move x* up
	sta PStack,x
	dex
	dex
	cpx W
	bne @12
	sty PStack,x	; store xu

@99:	rts
  .endif

;==============================================================================
; SECTION 2: RETURN STACK


 Header ">R",0 ; ( x -- ) ( R: -- x )  push to return stack
  ; https://forth-standard.org/standard/core/toR
ToR:
	LDA PStack,X	; Pop from parameter stack
	INX
	INX
	ply		; save rts addr
	PHA		; Push onto return stack
	phy		; put rts addr back
	rts


 Header "2>R",0 ; ( x1 x2 -- ) ( R: -- x1 x2 )  move cell pair to return stack
  ; https://forth-standard.org/standard/core/TwotoR
TwoToR:
	ply		; save rts addr
	lda PStack+2,x	; get x1
	pha
	lda PStack+0,x	; get x2
	pha
	phy		; put rts addr back
	jmp TwoDrop	; 2Drop


 Header "R>",0 ; ( -- x ) ( R: x -- )  pop from return stack
  ; https://forth-standard.org/standard/core/Rfrom
RFrom:
	ply		; save rts addr
	PLA		; Pop from return stack
	phy		; put rts addr back
	jmp PushA	; Push onto parameter stack


 Header "2R>",0 ; ( R: x1 x2 -- ) ( -- x1 x2 )  Pop 2 cells from the return stack
  ; https://forth-standard.org/standard/core/TwoRfrom
TwoRFrom:
	dex		; alloc space
	dex
	dex
	dex
	ply		; save rts addr
	pla		; do x2
	sta PStack+0,x
	pla		; do x1
	sta PStack+2,x
	phy		; restore rts addr
	rts


 Header "R@",0 ; ( R: x -- x ) ( -- x )  get a copy of top of return stack
  ; https://forth-standard.org/standard/core/RFetch
RFetch:
	lda 3,s
	jmp PushA


 Header "2R@",0 ; ( R: x1 x2 -- x1 x2 ) ( -- x1 x2 )  Get top 2 return stack cells
  ; https://forth-standard.org/standard/core/TwoRFetch
TwoRFetch:
	lda 5,s
	PUSHA
	lda 3,s
	jmp PushA

;==============================================================================
; SECTION 5: LOGIC

 Header "True",0 ; ( -- f )
  ; https://forth-standard.org/standard/core/TRUE
True:
	lda #$ffff
	jmp PushA


 Header "False",0 ; ( -- f )
  ; https://forth-standard.org/standard/core/FALSE
False:
	lda #0
	jmp PushA


 Header "Invert",0 ; ( x1 -- x2 )  bitwise invert
  ; https://forth-standard.org/standard/core/INVERT
Invert:
	LDA PStack,X
	EOR #$FFFF
	STA PStack,X
	rts


 Header "And",0 ; ( x1 x2 -- x3 )  bitwise And
  ; https://forth-standard.org/standard/core/AND
  ; Modified name to not conflict with opcode name.
And1:
	LDA PStack+2,X
	AND PStack+0,X
	bra NipA


 Header "Or",0 ; ( x1 x2 -- x3 )  bitwise Or
  ; https://forth-standard.org/standard/core/OR
  ; Modified name to not conflict with opcode name.
Or1:
	LDA PStack+2,X
	ORA PStack+0,X
	bra NipA


 Header "Xor",0 ; ( x1 x2 -- x3 )  bitwise XOr
  ; https://forth-standard.org/standard/core/XOR
  ; Modified name to not conflict with opcode name.
Xor1:
	LDA PStack+2,X
	EOR PStack+0,X
	bra NipA


;==============================================================================
; SECTION 3: ARITHMETIC


 Header "Max",0 ; ( n1 n2 -- nmax )  signed
  ; https://forth-standard.org/standard/core/MAX
Max:
	sec
	lda PStack+0,x	; n1
	sbc PStack+2,x	; n2
	bvs Min5
Max5:	bpl Nip
	inx		; Drop
	inx
	rts


 Header "Min",0 ; ( n1 n2 -- nmin )  signed
  ; https://forth-standard.org/standard/core/MIN
Min:
	sec
	lda PStack+0,x	; n1
	sbc PStack+2,x	; n2
	bvs Max5
Min5:	bmi Nip
	inx		; Drop
	inx
	rts


 Header "Nip",0 ; ( x1 x2 -- x2 )
  ; https://forth-standard.org/standard/core/NIP
Nip:
	LDA PStack,X	; x2
NipA:	INX
	INX
	STA PStack,X	; Overwrite x1
	rts


 Header "+",0 ; ( x1 x2 -- x3 )  Add
  ; https://forth-standard.org/standard/core/Plus
Plus:
	clc
	lda PStack+2,x	; get a
	adc PStack+0,x	; add b
	bra NipA


 Header "-",0 ; ( x1 x2 -- x3 )  Subtract
  ; https://forth-standard.org/standard/core/Minus
Minus:
	sec
	lda PStack+2,x	; get x1
	sbc PStack+0,x	; subtract x2
	bra NipA


 Header "Negate",0 ; ( n1 -- n2 )
  ; https://forth-standard.org/standard/core/NEGATE
Negate:
	LDA PStack,X
Negat3:	EOR #$FFFF
	INC A
	STA PStack,X
	rts


 Header "Abs",0 ; ( n -- u )  Absolute value
  ; https://forth-standard.org/standard/core/ABS
Abs:
	LDA PStack,X
	bmi Negat3
	rts


 Header "M+",0 ; ( d1 n -- d2 )  Add signed single to double
  ; https://forth-standard.org/standard/double/MPlus
MPlus:
	jsr SToD
	bra DPlus


 Header "D+",0 ; ( d1 d2 -- d3 ) double precision add
  ; https://forth-standard.org/standard/double/DPlus
DPlus:
	clc
	lda PStack+6,x	; get d1.lo
	adc PStack+2,x	; add to d2.lo
	sta PStack+6,x
	lda PStack+4,x	; get d1.hi
	adc PStack+0,x	; add to d2.hi
	sta PStack+4,x
	jmp TwoDrop


 Header "D-",0 ; ( d1 d2 -- d3 ) double precision subtract
  ; https://forth-standard.org/standard/double/DMinus
DMinus:
	jsr DNegate
	bra DPlus


 Header "DNegate",0 ; ( d1 -- d2 )  double negate
  ; https://forth-standard.org/standard/double/DNEGATE
DNegate:
	sec
	lda #0		; do d1.lo
	sbc PStack+2,x
	sta PStack+2,x
	lda #0		; do d1.hi
	sbc PStack+0,x
	sta PStack+0,x
	rts


 Header "DAbs",0 ; ( d -- ud )  ud is the absolute value of d
  ; https://forth-standard.org/standard/double/DABS
DAbs:
	lda PStack+0,x	; test d sign
	bmi DNegate
	rts


 Header "1+",F_Inline ; ( n1 -- n2 )  increment
  ; https://forth-standard.org/standard/core/OnePlus
OnePlus:
	inc PStack,x
	INL_End


 Header "2+",0 ; ( n1 -- n2 )  increment by 2
TwoPlus:
	inc PStack,x
	bra OnePlus


 Header "1-",F_Inline ; ( n1 -- n2 )  decrement
  ; https://forth-standard.org/standard/core/OneMinus
OneMinus:
	dec PStack,x
	INL_End


 Header "2*",F_Inline ; ( n -- n*2 )  shift left
  ; https://forth-standard.org/standard/core/TwoTimes
TwoStar:
	asl PStack,x
	INL_End


 Header "D2*",F_Inline ; ( d1 -- d2 )  double shift left
  ; https://forth-standard.org/standard/double/DTwoTimes
DTwoStar:
	asl PStack+2,x
	rol PStack+0,x
	INL_End


 Header "2/",0 ; ( n1 -- n2 ) signed shift right
  ; https://forth-standard.org/standard/core/TwoDiv
TwoSlash:
	LDA PStack,x
	asl a		; Arithmetic shift right: preserve sign bit
	ROR PStack,x	; Shift right, sign bit from carry
	rts


 Header "D2/",0 ; ( d1 -- d2 ) double signed shift right
  ; https://forth-standard.org/standard/double/DTwoDiv
DTwoSlash:
	jsr TwoSlash
	ror PStack+2,x
	rts


 Header "U2/",F_Inline ; ( u1 -- u2 ) unsigned shift right
UTwoSlash:
	lsr PStack,x
	INL_End


 Header "DU2/",F_Inline ; ( ud1 -- ud2 ) double unsigned shift right
DUTwoSlash:
	lsr PStack+0,x
	ror PStack+2,x
	INL_End


 Header "LShift",0 ; ( n1 u -- n2 )  shift left by u
  ; https://forth-standard.org/standard/core/LSHIFT
LShift:	ldy PStack+0,x	; get u (shift count)
	beq @done
@loop:	asl PStack+2,x	; shift a
	dey
	bne @loop
@done:	inx		; Drop u
	inx
	rts


 Header "RShift",0 ; ( n1 u -- n2 ) logical shift right by u
  ; https://forth-standard.org/standard/core/RSHIFT
RShift:	ldy PStack+0,x	; get u (shift count)
	beq @done
@loop:	lsr PStack+2,x	; shift a
	dey
	bne @loop
@done:	inx		; Drop u
	inx
	rts


  .if 0
 Header "><",0 ; ( x1 -- x2 )  Swap byte order in cell
SwapB:
	lda PStack,x
	xba
	sta PStack,x
	rts
   .endif


 Header "S>D",0 ; ( n -- d )  Convert the signed number n to the double-cell number d
  ; https://forth-standard.org/standard/core/StoD
SToD:
	lda PStack,x	; test sign of n
	bpl UToD
	jmp True	; pad with -1


 Header "U>D",0 ; ( u -- ud )  Convert the unsigned number n to the double-cell number d
UToD:	jmp False	; pad with 0


 Header "D>S",F_Inline ; ( xd -- x )  Convert double to single
  ; https://forth-standard.org/standard/double/DtoS
DToS:
	inx		; drop d.hi
	inx
	INL_End


 Header "UM*",0 ; ( u1 u2 -- ud ) unsigned 16x16 -> 32-bit result
  ; Result: TOS = high cell, NOS = low cell
  ; https://forth-standard.org/standard/core/UMTimes
UMStar:
	lda #0		; Clear result.hi
	ldy #16		; for 16 bits
@loop:	ror PStack+2,x	;   shift multiplier & result.lo
	bcc @skip	;   if bit set
	clc		;     add multiplicand to result.hi
	adc PStack+0,x
@skip:	ror a		;   shift result.hi
	dey		;  next bit
	bne @loop
	ror PStack+2,x	; finish result.lo
	sta PStack+0,x	; save result.hi
	rts


 Header "UN*",0 ; ( ud1 u2 -- ud3 ) unsigned 32x16 -> 32-bit result
UNStar:
 .if 1
	lda PStack,x	; Dup
	pha		; >R
	jsr UMStar	; UM*
	inx		; Drop
	inx
	jsr Swap	; Swap
	pla		; R>
	PUSHA
	jsr UMStar	; UM*
	jsr Rot		; Rot
	jmp Plus	; +
  .else
	lda PStack+4,x	; Temp= ud1
	sta Temp0
	lda PStack+2,x
	sta Temp2
	stz PStack+4,x	; ud3= 0
	stz PStack+2,x
	bra @18

@11:	clc		; ud3+= Temp
	lda Temp0
	adc PStack+4,x
	sta PStack+4,x
	lda Temp2
	adc PStack+2,x
	sta PStack+2,x
@15:	asl Temp0	; Temp*=2
	rol Temp2
@18:	lsr PStack+0,x	; shift u2
	bcs @11
	bne @15

	inx		; drop u2
	inx
	rts
  .endif


 Header "M*",0 ; ( a b -- dc ) 16x16 -> 32 signed
  ; https://forth-standard.org/standard/core/MTimes
MStar:
	lda PStack+0,x	; save result sign
	eor PStack+2,x
	php
	jsr Abs		; abs(b)
	jsr Swap	; abs(a)
	jsr Abs
	jsr UMStar	; multiply
	plp		; fix result sign
	bpl @9
	jmp DNegate
@9:	rts


 Header "*",0 ; ( a b -- a*b ) 16x16 -> 16 (low word)
  ; https://forth-standard.org/standard/core/Times
Star:
	jsr UMStar
	inx		; drop hi word
	inx
	rts


 Header "UN/Mod",0 ; ( ud u -- udq ur )  unsigned 32/16 -> 32 quotient, 16 remainder
UNSlashMod:
	lda #0		; dividend.hi
	asl PStack+4,x	; shift dividend.lo & mid
	rol PStack+2,x
	ldy #32		; for 32 bits
@11:    rol a		;   Shift dividend.hi
	bcs @24
	cmp PStack+0,x	;   will it fit?
	bcc @27
@24:	sbc PStack+0,x	;     subtract divisor
	sec
@27:
	rol PStack+4,x	;   shift dividend & quotient lo & mid
	rol PStack+2,x
	dey
	bne @11		;  next bit
	sta PStack+0,x	; save remainder
	rts


 Header "UM/Mod",0 ; ( ud u -- ur uq ) unsigned 32/16 -> 16 remainder, 16 quotient
  ; Stack: NOS_HI=ud_high NOS=ud_low TOS=u (divisor)
  ; Quotient overflow gives strange results.
  ; This is a standard 32/16 non-restoring division
  ; https://forth-standard.org/standard/core/UMDivMOD
UMSlashMod:
	jsr UNSlashMod
	jsr Nip
	jmp Swap


 Header "SM/Rem",0 ; ( d1 n1 -- n_remainder n_quotient )  Symmetric signed division
  ; Divide d_dividend by n_divisor, giving the symmetric quotient n_quotient and
  ; the remainder n_remainder. Input and output stack arguments are signed.
  ; An ambiguous condition exists if n_divisor is zero or if the quotient
  ; lies outside the range of a single-cell signed integer.
  ; http://forth-standard.org/standard/core/SMDivREM
SMSRem:
	jsr ToR		; save divisor
	lda PStack+0,x	; save dividend sign
	pha
	jsr DAbs	; make dividend positive
	lda 3,s
	jsr PushA	; get abs(divisor)
	jsr Abs
	jsr UMSlashMod	; unsigned divide
	lda 3,s		; correct quotient sign
	eor 1,s
	bpl @5
	jsr Negate
@5:
	pla		; correct remainder sign
	bpl @6
	lda #0
	sec
	sbc PStack+2,x
	sta PStack+2,x
@6:
	pla		; drop saved divisor
	rts


 Header "/Mod",0 ; ( n1 n2 -- rem quot ) signed division
  ; Divide n1 by n2, giving the single-cell remainder n3 and the single-cell quotient n4. An
  ; ambiguous condition exists if n2 is zero. If n1 and n2 differ in sign, the implementation defined
  ; result returned will be the same as that returned by either the phrase >R S>D R>
  ; FM/MOD or the phrase >R S>D R> SM/REM.
  ; https://forth-standard.org/standard/core/DivMOD
SlashMod:
	jsr ToR
	jsr SToD
	jsr RFrom
	bra SMSRem


 Header "/",0 ; ( n1 n2 -- quot ) signed division
  ; https://forth-standard.org/standard/core/Div
Slash:
	jsr SlashMod
	jmp Nip		; remainder



 Header "Mod",0 ; ( n1 n2 -- rem ) signed remainder
  ; https://forth-standard.org/standard/core/MOD
Mod:
	jsr SlashMod
	inx		; Drop quotient
	inx
	rts

 Header "Search-WordList",0 ; ( caddr u wid -- 0 | xt 1 | xt -1)  "Search for a word in a wordlist"
  ; Find the definition named in the string.
  ; If the definition is not found, return zero.
  ; If the definition is found, return its execution token xt.
  ; If the definition is immediate, also return one (1), otherwise also return minus-one (-1).
  ; https://forth-standard.org/standard/search/SEARCH_WORDLIST
Search_WordList:
@HdrPtr	= Temp0  ; ptr to current word Header (-@Off)
@Quick	= Temp2  ; pattern last char & length
@StartY = Temp4  ; char compare start offset
@PatPtr = Temp6  ; ptr to pattern addr
@Off = $83 ; Chosen so 8-bit Y turns negative as we step off the end of the name chars.
		; @Off is used because the 65816 doesn't do negative indexes.

	lda #@Off-3-1	; char compare start offset
	sec
	sbc PStack+2,x
	sta @StartY

	lda PStack+4,x	; ptr to pattern
	clc
	sbc @StartY
	sta @PatPtr

	lda PStack+2-1,x ; get pattern name length in hi byte
	sep #P_M
	ldy #@Off-4	; get pattern name last char in lo byte
	lda (@PatPtr),y
	rep #P_M
	sta @Quick	; save @Quick compare value

	lda PStack+0,x	; A= wid (wordlist head pointer)
	bra @WCur

@WNxt:	rep #P_M|P_X
	ldy @HdrPtr	; Y= header pointer
@WNxtY:	lda a:@Off-2,y	; follow link to next header
@WCur:	beq @nfnd	;   end of chain?
	sec		; apply offset for indexing
	sbc #@Off
	tay
	lda a:@Off-4,y	; check length & smudge & last char
	eor @Quick
	and #F_LengthMask<<8+F_Hidden<<8+$df
	bne @WNxtY

	sty @HdrPtr	; save header pointer
	sep #P_M|P_X
	ldy @StartY	; for each char in name
@NNxt:	iny
	bmi @fnd
	lda (@HdrPtr),y	;   compare char
	eor (@PatPtr),y
	beq @NNxt	;   exact match?
	and #$df	;   more difference than a case mismatch?
	bne @WNxt
	lda (@HdrPtr),y	;   is it a letter?
	and #$df
	sec
	sbc #'A'
	cmp #'Z'-'A'+1
	bcc @NNxt
	bra @WNxt

@fnd:			;found the word
	rep #P_M|P_X
	ldy @HdrPtr	; Y= header pointer
	lda a:@Off-3-1,y ; set flag from F_Immediate
	ora #$7fff
	bmi @fnd2
	lda #1
@fnd2:	sta PStack+2,x
	tya		; set xt
	clc
	adc #@Off
	sta PStack+4,x
	bra @99

@nfnd:			;word not found
	stz PStack+4,x	; set flag
	inx		; drop len
	inx
@99:	inx		; drop wid
	inx
	rts


Search_Latest:

	jsr PushA
	jsr Search_WordList
	inx		; pop flag
	inx
	lda PStack-2,x	; set CPU flags
	rts

FSEmptyP1: lda #U_FSEmptyP1

	ldy U_ParseLen
	jmp PushYA

 Header "Count",0 ; ( addr -- addr+1 len ) counted string to addr/len
  ; https://forth-standard.org/standard/core/COUNT
Count:
	lda (PStack,x)	; get length byte
	and #$ff
	inc PStack,x

 .if 0
 Header "Bounds",0 ; ( addr u -- addr+u addr ) Prepare address for looping
  ; http://www.complang.tuwien.ac.at/forth/gforth/Docs-html/Memory-Blocks.html
Bounds:
	lda PStack+0,x

 Header "Compare",0 ; ( c-addr1 u1 c-addr2 u2 -- n )  Compare 2 strings
  ; https://forth-standard.org/standard/string/COMPARE
Compare:
@adr2	= Temp0	; addr of string2
@adr1	= Temp2	; addr of string1
@Len	= Temp4	; min(length of string1, length of string2)
	lda PStack+2,x	; get c-addr2
	sta @adr2
	lda PStack+6,x	; get c-addr1
	sta @adr1
	lda PStack+4,x	; get min(u1,u2)
	cmp PStack+0,x
	bcc @1
	lda PStack+0,x
@1:	sta @Len
	sep #P_M
	ldy #$ffff	; for each char
@2:	iny
	cpy @Len	;   out of chars?
	bcs @Out
	lda (@adr1),y	;   compare char
	cmp (@adr2),y
	beq @2
@6:	rol a		; b1,b0 = carry, 0
	asl a
	bra @8

@Out:	rep #P_M
	lda PStack+0,x
	cmp PStack+4,x	; which string is used up?
	bne @6
	lda #1		; return equal
@8:	rep #P_M
	and #3
	dea
	sta PStack+6,x	; push n
	jmp ThreeDrop

 Header "(",F_Immediate ; (  ccc(paren)  -- )  Comment
  ; https://forth-standard.org/standard/file/p
  ; http://forth-standard.org/standard/core/p
Comment:
	lda #')'
	jsr ParseA

 Header "\",F_Immediate ; ( -- )  Eat remainder of parse line as a comment
  ; https://forth-standard.org/standard/core/bs
BSlash:
	lda U_ParseLen
	sta U_ToIn
	rts


 Header "Parse",0 ; ( "name" c -- addr u )  Parse input with delimiter character
  ; https://forth-standard.org/standard/core/PARSE
  ; Find word in input string delimited by character given. Do not skip leading delimiters
  ; -- this is the main difference to PARSE-NAME.
  ; ANSI discussion: http://www.forth200x.org/documents/html3/rationale.html#rat:core:PARSE
Parse:	bra Parse2

ParseQuote: lda #'"'
ParseA:	jsr PushA
Parse2:	dex			; alloc 2nd parm cell
	dex
	sep #P_M

	ldy U_ToIn
	phy			;save start index

	dey			;span word
@22:	lda PStack+2,x		; load delimiter
@21:	iny			;  step to next char
	cpy U_ParseLen		;  at end-of-string?
	bcs @28
	cmp (U_ParseBuf),y	;  delimiter?
	beq @Found
	cmp #' '		; we're searching for whitespace?
	bne @21
	lda (U_ParseBuf),y	; is it whitespace?
	cmp #' '
	bcs @22

@Found:	iny			;  skip delimiter
	sty U_ToIn
	dey
	bra @29
@28:	sty U_ToIn
@29:
	rep #P_M

	tya		; calc length
	sec
	sbc 1,s
	sta PStack+0,x

	pla		; calc start addr
	clc
	adc U_ParseBuf
	sta PStack+2,x

	rts


 Header "Parse-Name",0 ; (  (spaces)name(space)  -- c-addr u )  Skip leading spaces. Parse space delimited name.
  ; https://forth-standard.org/standard/core/PARSE-NAME
  ; c-addr is the address of the selected string within the input buffer and u is its length in
  ; characters. If the parse area is empty or contains only white space, the resulting string
  ; has length zero.
Parse_Name:
	jsr SkipWhite		;skip leading whitespace
	lda #' '
	jsr ParseA		;get the name
;	bra SkipWhite		;skip trailing whitespace


; Header "SkipWhite",0 ; ( -- ) skip whitespace on source
SkipWhite:
	sep #P_M
	ldy U_ToIn
	dey
@11:	iny
	cpy U_ParseLen
	bcs @19
	lda (U_ParseBuf),y
	cmp #' '+1
	bcc @11
@19:	sty U_ToIn
	rep #P_M
	rts


 Header ">Number",0 ; ( ud1 c-addr1 u1 -- ud2 c-addr2 u2 )
  ; https://forth-standard.org/standard/core/toNUMBER
ToNumber:
	jsr TwoSwap	; ( c-addr1 u1 ud1 )
	lda PStack+4,x
	bra @18

@11:	lda (PStack+6,x) ;   get char
	and #$ff
	sec		;   get ASCII digit value
	sbc #'0'
	bcc @done
	cmp #10
	bcc @ok
	sbc #7
	and #$df
	cmp #10
	bcc @done
@ok:	cmp U_Base
	bcs @done
	pha		;   save digit
	lda U_Base	;   ud1*= Base
	jsr PushA
	jsr UNStar
	jsr RFrom	;   ud1+= digit
	jsr MPlus
	inc PStack+6,x	;   eat the char
	dec PStack+4,x
@18:	bne @11
@done:	jmp TwoSwap


 Header "Number",0 ; ( adr len -- 0 ) or ( adr len -- n -1 ) or ( adr len -- d -2 ) or ( adr len -- fp -3 )
  ; Converts the text beginning at adr for len, with regard to Base, to a binary
  ; value that is single-length if no valid punctuation occurs, and
  ; double-length if valid punctuation does occur.  The string may contain
  ; a preceding negative sign.
  ; https://gforth.org/manual/Number-Conversion.html
Number:
@Base	= 3	; saved numeric base
@Neg	= 1	; negative sign found flag
	jsr TwoDup	; ( dValue adr len )
	stz PStack+4,x	; dValue= 0
	stz PStack+6,x
	pei (U_Base)	; @Base  remember base before override
	pea 0		; @Neg  start positive

	jsr @Get	; get 1st char
	bmi @err
	ldy #10		; decimal base?
	cmp #'&'
	beq @SetBase
	cmp #'#'
	beq @SetBase
	ldy #2		; binary base?
	cmp #'%'
	beq @SetBase
	ldy #16		; hexadecimal base?
	cmp #'$'
	bne @NotBase
@SetBase: sty U_Base	; set temp base
	jsr @Get	; get next char
	bmi @err
@NotBase:

	cmp #'+'	; plus sign?
	beq @SC8
	cmp #'-'	; minus sign?
	bne @NotSign
	sta @Neg,s	; set negative flag
@SC8:	jsr @Get	; get next char
	bmi @err
@NotSign:

	cmp #$27	; (apostrophe) character?
	bne @NotChar
	jsr @Get	;  get char
	bmi @err
	sta PStack+6,x
	jsr @Get	; get trailing "'"
	bmi @Sign
	cmp #$27
	beq @Sign
	dec PStack+2,x	; unget
	inc PStack+0,x
	bra @Sign
@NotChar:
	dec PStack+2,x	; unget
	inc PStack+0,x

	jsr ToNumber

@Sign:	lda 1,s		; apply sign
	beq @Sign9
	jsr TwoSwap
	jsr DNegate
	jsr TwoSwap
@Sign9:

	jsr @Get	; get next char
	bmi @single	;   end of string?
	cmp #'.'	; double indicator?
	bne @err
	jsr @Get	; should be no more chars
	bpl @err
	lda #.loword(-2) ; return double
	bra @e5

@err:	inx		; Drop dValue.lo
	inx
	lda #0		; return fail
	bra @e3

@single: lda #.loword(-1) ; return single flag
@e3:	inx		; drop dValue.hi
	inx
@e5:	inx		; drop len
	inx
	sta PStack+0,x
	ply		; RDrop @Neg
	ply		; restore from @Base
	sty U_Base
	rts

@Get: ; get next char in A, P_M= end of string
	dec PStack+0,x	; @Len--  more chars?
	bmi @Get9
	lda (PStack+2,x) ; get char
	inc PStack+2,x	; @Adr++
	and #$ff	; zero extend
@Get9:	rts

 Header "Quit",0 ; ( -- ) outer interpreter loop
  ; Resets return stack, reads and interprets input forever
  ; https://forth-standard.org/standard/core/QUIT
Quit:
	lda #RSP_Init+UserFirst		; Reset return stack
	tcs
	stz U_State			; STATE= interpret

@Loop:	lda #TIB_Base			; Use terminal buffer

	sta U_ParseBuf
	ldy #TIB_Size			; push buf length
	jsr PushYA
	jsr Accept			; Read line
	lda PStack,x
	inx
	inx
	sta U_ParseLen			; save line size
	jsr Space			; type a space after the line
	stz U_ToIn			; init parse offset
	jsr Interpret			; Interpret the input line
	lda U_State			; STATE == 0 (interpret mode)?
	bne @8				; Loop back (compiling: no prompt)
	jsr DotQuoteRun
	.word 3				; Print " ok"
	.byte " ok"
@8:	jsr CR
	bra @Loop

 Header "Interpret",0 ; ( -- ) parse and execute/compile words from input
Interpret:
@next_word:
	jsr Parse_Name		; Parse next space-delimited word
	lda PStack,x
	beq @done		; no more words in source?

	jsr TwoDup		; ( caddr len caddr len )
	jsr Search_Latest	; search for the word in dictionary
	beq @not_found		; result: 0=not found, 1=normal, -1=immediate
	bmi @exec_word

	lda U_State
	beq @exec_word

	; Compile the word: , the xt
	jsr CompileComma
	jsr TwoDrop		; 2Drop extra string
	bra @next_word

@exec_word: ; Execute: load xt, jump through code field
	jsr Nip
	jsr Nip			; discard extra string copy
	jsr Execute		; run the xt
	bra @next_word

@not_found: ; not a name, try a number
	jsr TwoDup		; ( caddr len caddr len )
	jsr Number		; Try to convert as number
	lda PStack,x
	beq @Err		; ( adr len -- 0 ) or ( adr len -- n -1 ) or ( adr len -- d -2 )
	inx
	inx
	ina
	beq @Single

	; it's a double
	ldy U_State
	bne @DLit
	jsr TwoNip		; discard string, keep d
	bra @next_word

@DLit:	jsr TwoLiteral
	bra @nf7

@Single: ldy U_State
	bne @SLit
	jsr Nip
	jsr Nip			; discard string, keep n
	bra @next_word

@SLit:	jsr Literal		; compile n
@nf7:	jsr TwoDrop		; 2Drop string
	bra @next_word


@Err:	inx			; number flag
	inx
	jsr Type		; string
	jsr DotQuoteRun
	.word 11
	.byte " not found "
	jsr Abort
	
@done:	jmp TwoDrop		; discard empty parse string


 Header "[",F_Immediate ; ( -- )  switch State to interpret

OpenBracket:

	rts


 Header "]",0 ; ( -- )  switch State to compile
 ; https://forth-standard.org/standard/right-bracket
CloseBracket:

	rts

 Header "Dump",0 ; ( caddr len -- )  Dump memory in hex
  ; https://forth-standard.org/standard/tools/DUMP
Dump:

@line:	pha

	jsr DotHex

	inx
